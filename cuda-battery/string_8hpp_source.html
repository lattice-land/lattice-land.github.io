25 /** Allocate a string of size `n` using `allocator`. */
26 CUDA string(size_t n, const allocator_type& alloc = allocator_type()):
27 data_(n+1, alloc) /* +1 for null-termination */
28 {
29 // GCC/Clang may emit -Wstringop-overflow when accessing data_[n]
30 // even though data_ has size n+1. The conditional access ensures
31 // that the compiler can verify the write is safe.

33 // In case the user modifies the string.

36 data_[0] = '\0';
37 }

79 template <class IntegerType>
80 CUDA NI static this_type from_int(IntegerType x, const allocator_type& alloc = allocator_type()) {
81 if(x == 0) { return this_type("0", alloc); }
82 size_t s = 0;
83 bool neg = x < IntegerType{0};
84 if(neg) {
85 x = -x;
86 s++;
87 }
88 for(size_t y = x; y > 0; y = y / 10, ++s) {}
89 this_type buffer(s, alloc);
90 if(neg) {
91 buffer[0] = '-';
92 }
93 for(size_t i = s-1; x > 0; --i) {
94 buffer[i] = '0' + (x % 10);
95 x = x / 10;
96 }
97 return std::move(buffer);
98 }

107 CUDA NI string<Allocator> concat(const char* lhs, size_t lhs_len, const char* rhs, size_t rhs_len, const Allocator& alloc) {
108 string<Allocator> res(lhs_len + rhs_len, alloc);
109 size_t k = 0;
110 for(size_t i = 0; i < lhs_len; ++i, ++k) { res[k] = lhs[i]; }
111 for(size_t i = 0; i < rhs_len; ++i, ++k) { res[k] = rhs[i]; }
112 return std::move(res);
113 }
